================Jenkins Pipeline -- Parallel Build and test========


pipeline{
  agent any
  stages{
     stage('Parallel Task'){
         parallel{
            stage('build'){
	        steps{
                  sh 'maven clean compile'
		}
	    }
	    stage('Test'){
	       steps{
                 sh 'mvn test'
	       }

	    }

	 }
     }
  }
}


=============== Gitlab CI--Parallel Build==============

stages:
  - build-test

maven-build:
  stage: build-test
  image: maven:3.8.1-jdk-17
  script:
    - git clone: https://github.com/Pathnex/sample-java-app.git
    - cd sample-java-app
    - mvn clean package

maven-test:
   stage:build-test
   image: maven:3.8.1-jdk-17
   script:
     - cd sample-java-app
     - mvn test
   parallel: 1
