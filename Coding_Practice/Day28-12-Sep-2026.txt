============ Jenkins Autimatic Rollback ============

pipeline{

   agent any
    
   environment{
      INSTITUTE_NAME = "Pathnex"

      TEAM = "Devops"

      ENV = "prod"
      
   }

   stages{
      stage('Deploy'){
            steps{
               script{
                  try{
                      sh 'echo "deploying  $INSTITUTE_NAME app in $ENV for $TEAM"'
                        sh 'exit 1'
                    } catch (Exception e) {
                        echo "Deployment failed. Rolling back..."

                        sh 'kubectl rollout undo deployment/my-app'
                    }

		  }
	       }
	    }
      }

   }
}
