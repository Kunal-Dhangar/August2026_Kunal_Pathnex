==================Jenkins-Pipeline==============

pipeline{
  agent any

  stages{
    stage('Checkout'){
       steps{
          git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
       }
    }
    stage('Build'){
       steps{
         sh '''
	   mvn clean compile || {echp "Compile Failed"; exit 1;}
	   '''
       }
    }
    stage('Test'){
       steps{
           sh '''
	     mvn test || {echo "test Failed"; exit 1;}
	     '''
       }
    }
  }
}
