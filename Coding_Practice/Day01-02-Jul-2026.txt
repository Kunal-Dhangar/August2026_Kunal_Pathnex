##### Ansible Task- Install Nginx on Pathnex Server


- name :Install Nginx on Pathnx Server
  hosts:all
  become:yes 


  tasks:
   - name: Install nginx
     yum:
       name: nginx
       state: present

######## Teraform Task -Create Ec2 (c5.xlarge)

provider "aws"{
    region= "us-east-1"
   }

resource "aws_instance" "PathnexEC2" {
   ami    ="ami-0abcd1234abcd1234"
   instance_type= "c5.xlarge"

   tags= {

     Name:"Pathnex-EC2"
     }
}  


######### Kubernetes Task- Create Ngnox Pod

apiVersion: v1
kind: pod
metadata:
  name:pathnex-pod
spec:
   containers: 
    - name: web
      image: nginx:latest
      ports:
       - containers:80

#########Jenkins Pipeline - Checkout Git=============

pipeline{

   agent any
    stages{
     stage('Checkout'){
        steps{
	    git branch: 'main',url:'https://github.com/Pathnex/sample-java-app.git'
        }
      }
      stage('List Files'){
         step{
	    sh 'ls -la
	 }
       }
    
   }
}
